--------
  solves problem of service discovery 

  frontend  ---> service (provide labels)---> backend


Kubernetes yaml : Deployment resource
=====================================

 apiVersion: V1
 kind: Deployment   #name of the deployment
 metadata:
   name:
   labels:          #related to project and service information  - not important for service discovery 
     app.kubernetes.io/team-name: ad-service       #for reperesent the reservice/organizatoin 

 spec:
   replicas:         #manditay fields- Replica sets
   template:        # pod related configuration 
     metadata:  
       label:        #service discovery 
         app.kubernets.io/name:ad-service    #service name 
         app.kubernets.io/name:team-name

     spec:
      serviceaccountnamr: 
      containers:
        - name: adservice
          image:              #image name from registry
          ports:
          env:
          volume:



 Kubernetes yaml : Service resource
 ==================================

 apiVersion: V1
 kind: service 
 metadata:
   name:     
   labels:
      
 spec:
  type: clusetrIP/node port/load balancer              #internal and external network configuration 
  ports:                                    
    - port: 8080         # not a container port. it is  FQDN:port   - fully qualified domain name 
      targetport: 8080    # service pod port 

  selector:   #label- name of current pod name

service type:
=============
  services:  #can configure by 
    - clusetrIP    # only allows with in cluster configuration between the node in VPC 
    - Node port    # allows access within the VPC 
    - LB           # API  ---> CCM (generate FQDN) ---> AWS ---> LB ---> assign public IP
               ##CCM - Cloud control manager
